find_program(PARTICLESATURN_XCRUN xcrun REQUIRED)

set(PARTICLE_ABI_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/generated/shaders")
set(PARTICLESATURN_METAL_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/generated/metal")
set(PARTICLESATURN_METAL_AIR "${PARTICLESATURN_METAL_OUTPUT_DIRECTORY}/ParticleKernels.air")
set(PARTICLESATURN_METAL_LIBRARY "${PARTICLESATURN_METAL_OUTPUT_DIRECTORY}/ParticleKernels.metallib")

add_custom_command(
    OUTPUT "${PARTICLESATURN_METAL_LIBRARY}"
    BYPRODUCTS "${PARTICLESATURN_METAL_AIR}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${PARTICLESATURN_METAL_OUTPUT_DIRECTORY}"
    COMMAND "${PARTICLESATURN_XCRUN}" -sdk macosx metal -I "${PARTICLE_ABI_OUTPUT_DIRECTORY}" -c "${CMAKE_CURRENT_LIST_DIR}/ParticleKernels.metal" -o "${PARTICLESATURN_METAL_AIR}"
    COMMAND "${PARTICLESATURN_XCRUN}" -sdk macosx metallib "${PARTICLESATURN_METAL_AIR}" -o "${PARTICLESATURN_METAL_LIBRARY}"
    DEPENDS "${CMAKE_CURRENT_LIST_DIR}/ParticleKernels.metal" "${PARTICLE_ABI_MSL}"
    VERBATIM
)

add_custom_target(ParticleSaturnMetalShaders ALL DEPENDS "${PARTICLESATURN_METAL_LIBRARY}")
add_dependencies(ParticleSaturnMetalShaders ParticleSaturnShaderAbi)
